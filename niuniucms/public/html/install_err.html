安装说明

方式1、
cd 

wget https://www.php.net/distributions/php-.tar.gz

tar -xf php-.tar.gz
cd php-/ext/ffi
bin/phpize
./configure --with-php-config=bin/php-config
make
make install
sed -i 's/^;extension=ffi/extension=ffi/' etc/php.ini
sed -i 's/^;ffi\.enable=preload/ffi.enable=true/' etc/php.ini
/etc/init.d/php-fpm-74 restart
#自行重启php服务
